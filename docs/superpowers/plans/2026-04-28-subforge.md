#include "config.h"
#include <fstream>
#include <sstream>
#include <map>
#include <cstdlib>

namespace subforge {

Config& Config::instance() {
    static Config instance;
    return instance;
}

Config::Config() {
    load_env_file();
}

void Config::load_env_file(const std::string& path) {
    std::ifstream file(path);
    if (!file.is_open()) return;

    std::string line;
    while (std::getline(file, line)) {
        if (line.empty() || line[0] == '#') continue;

        auto pos = line.find('=');
        if (pos == std::string::npos) continue;

        std::string key = line.substr(0, pos);
        std::string value = line.substr(pos + 1);

        // 去除引号
        if (value.size() >= 2 && value.front() == '"' && value.back() == '"') {
            value = value.substr(1, value.size() - 2);
        }

        values_[key] = value;
    }
}

std::string Config::get(const std::string& key, const std::string& default_val) const {
    auto it = values_.find(key);
    if (it != values_.end()) return it->second;

    // 回退到环境变量
    const char* env = std::getenv(key.c_str());
    if (env) return env;

    return default_val;
}

int Config::get_int(const std::string& key, int default_val) const {
    std::string val = get(key);
    if (val.empty()) return default_val;
    try { return std::stoi(val); }
    catch (...) { return default_val; }
}

std::string Config::whisper_api_key() const { return get("WHISPER_API_KEY"); }
std::string Config::server_host() const { return get("SERVER_HOST", "0.0.0.0"); }
int Config::server_port() const { return get_int("SERVER_PORT", 8080); }
std::string Config::ffmpeg_path() const { return get("FFMPEG_PATH", "/usr/bin/ffmpeg"); }
std::string Config::ffprobe_path() const { return get("FFPROBE_PATH", "/usr/bin/ffprobe"); }

#include <nlohmann/json.hpp>

using json = nlohmann::json;

namespace subforge {

AudioExtractor::AudioExtractor() {
    auto& config = Config::instance();
    ffmpeg_path_ = config.ffmpeg_path();
    ffprobe_path_ = config.ffprobe_path();
}

std::string AudioExtractor::run_ffmpeg(const std::string& args) {
    std::string cmd = ffmpeg_path_ + " " + args + " 2>&1";
    std::array<char, 128> buffer;
    std::string result;
    std::unique_ptr<FILE, decltype(&pclose)> pipe(popen(cmd.c_str(), "r"), pclose);
    if (!pipe) throw std::runtime_error("popen() failed!");
    while (fgets(buffer.data(), buffer.size(), pipe.get()) != nullptr) {
        result += buffer.data();
    }
    return result;
}

std::string AudioExtractor::run_ffprobe(const std::string& args) {
    std::string cmd = ffprobe_path_ + " " + args + " 2>&1";
    std::array<char, 128> buffer;
    std::string result;
    std::unique_ptr<FILE, decltype(&pclose)> pipe(popen(cmd.c_str(), "r"), pclose);
    if (!pipe) throw std::runtime_error("popen() failed!");
    while (fgets(buffer.data(), buffer.size(), pipe.get()) != nullptr) {
        result += buffer.data();
    }
    return result;
}

std::string AudioExtractor::extract(const std::string& video_path, const std::string& output_dir) {
    std::string audio_path = output_dir + "/audio.wav";
    std::string args = "-y -i \"" + video_path + "\" -vn -acodec pcm_s16le -ar 16000 -ac 1 \"" + audio_path + "\"";
    std::string output = run_ffmpeg(args);

    // 检查是否成功
    if (output.find("Error") != std::string::npos || output.find("Invalid") != std::string::npos) {
        throw std::runtime_error("Audio extraction failed: " + output);
    }

    return audio_path;
}

VideoInfo AudioExtractor::get_video_info(const std::string& video_path) {
    std::string args = "-v quiet -print_format json -show_format -show_streams \"" + video_path + "\"";
    std::string output = run_ffprobe(args);

    json probe = json::parse(output);

    VideoInfo info;
    info.file_path = video_path;

    // 解析视频流信息
    for (const auto& stream : probe["streams"]) {
        if (stream["codec_type"] == "video") {
            info.width = stream.value("width", 0);
            info.height = stream.value("height", 0);
            std::string fps_str = stream.value("r_frame_rate", "30/1");
            size_t slash_pos = fps_str.find('/');
            if (slash_pos != std::string::npos) {
                int num = std::stoi(fps_str.substr(0, slash_pos));
                int den = std::stoi(fps_str.substr(slash_pos + 1));
                info.fps = den > 0 ? num / den : 30;
            }
        }
    }

    // 解析格式信息
    info.duration = probe["format"].value("duration", 0.0);
    info.size_mb = probe["format"].value("size", 0.0) / (1024.0 * 1024.0);
    info.format = probe["format"].value("format_name", "unknown");

    return info;
}

} // namespace subforge

        throw std::runtime_error("WHISPER_API_KEY not configured");
    }

std::string Transcriber::call_whisper_api(const std::string& audio_path, const std::string& language) {
    httplib::Client cli("api.openai.com");
    cli.set_keep_alive(true);
    cli.set_connection_timeout(30);
    cli.set_read_timeout(300); // 5分钟超时
    cli.set_bearer_token_auth(api_key_.c_str());

    // 读取音频文件
    std::ifstream file(audio_path, std::ios::binary);
    if (!file.is_open()) {
        throw std::runtime_error("Cannot open audio file: " + audio_path);
    }
    std::string content((std::istreambuf_iterator<char>(file)), std::istreambuf_iterator<char>());

    // 构建 multipart 请求
    std::string boundary = "----SubForgeBoundary";
    std::string body;

    body += "--" + boundary + "\r\n";
    body += "Content-Disposition: form-data; name=\"file\"; filename=\"audio.wav\"\r\n";
    body += "Content-Type: audio/wav\r\n\r\n";
    body += content + "\r\n";

    body += "--" + boundary + "\r\n";
    body += "Content-Disposition: form-data; name=\"model\"\r\n\r\n";
    body += "whisper-1\r\n";

    if (language != "auto") {
        body += "--" + boundary + "\r\n";
        body += "Content-Disposition: form-data; name=\"language\"\r\n\r\n";
        body += language + "\r\n";
    }

    body += "--" + boundary + "\r\n";
    body += "Content-Disposition: form-data; name=\"response_format\"\r\n\r\n";
    body += "srt\r\n";

    body += "--" + boundary + "--\r\n";

    httplib::Headers headers = {
        {"Content-Type", "multipart/form-data; boundary=" + boundary}
    };

    auto res = cli.Post("/v1/audio/transcriptions", headers, body, "multipart/form-data; boundary=" + boundary);

    if (!res) {
        throw std::runtime_error("Whisper API request failed: no response");
    }

    if (res->status != 200) {
        throw std::runtime_error("Whisper API error: " + std::to_string(res->status) + " " + res->body);
    }

    return res->body;
}

std::vector<Subtitle> Transcriber::parse_srt(const std::string& srt_content) {
    std::vector<Subtitle> subtitles;
    std::istringstream stream(srt_content);
    std::string line;
    int id = 0;

    while (std::getline(stream, line)) {
        // 跳过空行和序号行
        if (line.empty() || std::isdigit(line[0])) continue;

        // 解析时间轴: 00:00:01,000 --> 00:00:04,000
        std::regex time_regex(R"((\d{2}):(\d{2}):(\d{2}),(\d{3})\s*-->\s*(\d{2}):(\d{2}):(\d{2}),(\d{3}))");
        std::smatch match;
        if (std::regex_search(line, match, time_regex)) {
            double start = std::stoi(match[1]) * 3600 + std::stoi(match[2]) * 60 +
                          std::stoi(match[3]) + std::stoi(match[4]) / 1000.0;
            double end = std::stoi(match[5]) * 3600 + std::stoi(match[6]) * 60 +
                        std::stoi(match[7]) + std::stoi(match[8]) / 1000.0;

            Subtitle sub;
            sub.id = ++id;
            sub.start_time = start;
            sub.end_time = end;

            // 读取字幕文本（可能多行）
            std::string text;
            while (std::getline(stream, line) && !line.empty()) {
                if (!text.empty()) text += "\n";
                text += line;
            }
            sub.text = text;
            subtitles.push_back(sub);
        }
    }

    return subtitles;
}

std::vector<Subtitle> Transcriber::transcribe(const std::string& audio_path, const std::string& language) {
    std::string srt = call_whisper_api(audio_path, language);
    return parse_srt(srt);
}

} // namespace subforge

std::string Exporter::generate_srt_file(const std::vector<Subtitle>& subtitles, const std::string& output_dir) {
    std::string path = output_dir + "/subtitles.srt";
    std::ofstream file(path);

    auto format_time = [](double seconds) -> std::string {
        int hours = static_cast<int>(seconds) / 3600;
        int minutes = (static_cast<int>(seconds) % 3600) / 60;
        int secs = static_cast<int>(seconds) % 60;
        int ms = static_cast<int>((seconds - static_cast<int>(seconds)) * 1000);
        std::ostringstream oss;
        oss << std::setfill('0') << std::setw(2) << hours << ":"
            << std::setw(2) << minutes << ":"
            << std::setw(2) << secs << ","
            << std::setw(3) << ms;
        return oss.str();
    };

    for (const auto& sub : subtitles) {
        file << sub.id << "\n";
        file << format_time(sub.start_time) << " --> " << format_time(sub.end_time) << "\n";
        file << sub.text << "\n\n";
    }

    return path;
}

std::string Exporter::generate_ass_file(const std::vector<Subtitle>& subtitles, const SubtitleStyle& style, const std::string& output_dir) {
    std::string path = output_dir + "/subtitles.ass";
    std::ofstream file(path);

    // ASS 文件头
    file << "[Script Info]\n";
    file << "Title: SubForge Subtitles\n";
    file << "ScriptType: v4.00+\n";
    file << "PlayResX: 1920\n";
    file << "PlayResY: 1080\n\n";

    file << "[V4+ Styles]\n";
    file << "Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding\n";

    // 颜色转换 (HEX -> ASS &HBBGGRR)
    auto hex_to_ass_color = [](const std::string& hex) -> std::string {
        if (hex.size() < 7) return "&H00FFFFFF";
        std::string r = hex.substr(1, 2);
        std::string g = hex.substr(3, 2);
        std::string b = hex.substr(5, 2);
        return "&H00" + b + g + r;
    };

    int alignment = 2; // bottom center
    if (style.position == "top_center") alignment = 8;
    else if (style.position == "center") alignment = 5;

    file << "Style: Default," << style.font_family << "," << style.font_size << ","
         << hex_to_ass_color(style.color) << ",&H000000FF," << hex_to_ass_color(style.outline_color) << ","
         << hex_to_ass_color(style.bg_color) << ","
         << (style.bold ? 1 : 0) << "," << (style.italic ? 1 : 0) << ",0,0,100,100,0,0,1,"
         << style.outline_width << ",0," << alignment << ",10,10,10,1\n\n";

    file << "[Events]\n";
    file << "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text\n";

    auto format_time_ass = [](double seconds) -> std::string {
        int hours = static_cast<int>(seconds) / 3600;
        int minutes = (static_cast<int>(seconds) % 3600) / 60;
        int secs = static_cast<int>(seconds) % 60;
        int cs = static_cast<int>((seconds - static_cast<int>(seconds)) * 100);
        std::ostringstream oss;
        oss << std::setfill('0') << std::setw(1) << hours << ":"
            << std::setw(2) << minutes << ":"
            << std::setw(2) << secs << "."
            << std::setw(2) << cs;
        return oss.str();
    };

    for (const auto& sub : subtitles) {
        file << "Dialogue: 0," << format_time_ass(sub.start_time) << "," << format_time_ass(sub.end_time)
             << ",Default,,0,0,0,," << sub.text << "\n";
    }

    return path;
}

std::string Exporter::build_filter_complex(const std::string& subtitle_file, const SubtitleStyle& style) {
    // 使用 subtitles filter 烧录字幕
    std::string filter = "subtitles='" + subtitle_file;

    // 如果是 ASS 文件，不需要 force_style
    if (subtitle_file.find(".ass") != std::string::npos) {
        filter += "'";
    } else {
        // SRT 文件，应用样式
        std::string ass_style = "force_style='";
        ass_style += "FontName=" + style.font_family + ",";
        ass_style += "FontSize=" + std::to_string(style.font_size) + ",";

        auto hex_to_ass_color = [](const std::string& hex) -> std::string {
            if (hex.size() < 7) return "&H00FFFFFF";
            std::string r = hex.substr(1, 2);
            std::string g = hex.substr(3, 2);
            std::string b = hex.substr(5, 2);
            return "&H" + b + g + r;
        };

        ass_style += "PrimaryColour=" + hex_to_ass_color(style.color) + ",";
        ass_style += "OutlineColour=" + hex_to_ass_color(style.outline_color) + ",";
        ass_style += "BackColour=" + hex_to_ass_color(style.bg_color) + ",";
        ass_style += "Outline=" + std::to_string(style.outline_width) + ",";

        int alignment = 2;
        if (style.position == "top_center") alignment = 8;
        else if (style.position == "center") alignment = 5;
        ass_style += "Alignment=" + std::to_string(alignment);

        ass_style += "'";
        filter += ":" + ass_style;
    }

    return filter;
}

std::string Exporter::get_resolution(const std::string& resolution, int orig_width, int orig_height) {
    if (resolution == "original") return "";
    if (resolution == "720p") return "-vf scale=1280:720";
    if (resolution == "1080p") return "-vf scale=1920:1080";
    return "";
}

std::string Exporter::get_codec(const std::string& codec) {
    if (codec == "h264") return "libx264";
    if (codec == "h265") return "libx265";
    return "libx264";
}

std::string Exporter::burn_subtitles(const std::string& video_path,
                                     const std::string& subtitle_file,
                                     const SubtitleStyle& style,
                                     const ExportOptions& options,
                                     const std::string& output_dir) {
    std::string output_path = output_dir + "/output_" + std::to_string(time(nullptr)) + ".mp4";

    std::string filter = build_filter_complex(subtitle_file, style);
    std::string codec = get_codec(options.codec);

    std::string cmd = ffmpeg_path_ + " -y -i \"" + video_path + "\"";
    cmd += " -vf \"" + filter + "\"";
    cmd += " -c:v " + codec;
    cmd += " -preset medium";
    cmd += " -c:a copy";
    cmd += " \"" + output_path + "\" 2>&1";

    std::array<char, 128> buffer;
    std::string result;
    std::unique_ptr<FILE, decltype(&pclose)> pipe(popen(cmd.c_str(), "r"), pclose);
    if (!pipe) throw std::runtime_error("popen() failed!");
    while (fgets(buffer.data(), buffer.size(), pipe.get()) != nullptr) {
        result += buffer.data();
    }

    if (result.find("Error") != std::string::npos) {
        throw std::runtime_error("FFmpeg export failed: " + result);
    }

    return output_path;
}

TaskManager& TaskManager::instance() {

    return instance;

std::string generate_uuid() {
    uuid_t uuid;
    uuid_generate(uuid);
    char buf[37];
    uuid_unparse(uuid, buf);
    return std::string(buf);
}

std::string TaskManager::create_task(std::function<void(TaskStatus&)> task_func) {
    std::lock_guard<std::mutex> lock(mutex_);
    std::string task_id = generate_uuid();

    TaskStatus status;
    status.task_id = task_id;
    status.progress = 0;
    status.status = "pending";
    tasks_[task_id] = status;

    // 在后台线程执行任务
    std::thread([this, task_id, task_func]() {
        update_task_progress(task_id, 0, "processing");
        try {
            TaskStatus& task = tasks_[task_id];
            task_func(task);
            update_task_progress(task_id, 100, "done");
        } catch (const std::exception& e) {
            std::lock_guard<std::mutex> lock(mutex_);
            tasks_[task_id].status = "error";
            tasks_[task_id].error_message = e.what();
        }
    }).detach();

    return task_id;
}

TaskStatus TaskManager::get_task_status(const std::string& task_id) {
    std::lock_guard<std::mutex> lock(mutex_);
    auto it = tasks_.find(task_id);
    if (it != tasks_.end()) return it->second;
    throw std::runtime_error("Task not found: " + task_id);
}

void TaskManager::update_task_progress(const std::string& task_id, int progress, const std::string& status) {
    std::lock_guard<std::mutex> lock(mutex_);
    auto it = tasks_.find(task_id);
    if (it != tasks_.end()) {
        it->second.progress = progress;
        it->second.status = status;
    }
}

int TimelineEditor::find_subtitle_at(int x) const {
            double time = x_to_time(x);
            for (int i = 0; i < subtitles_.size(); ++i) {
                if (time >= subtitles_[i].start_time && time <= subtitles_[i].end_time) {
                    return i;
                }
            }
            return -1;
        }

        void TimelineEditor::paintEvent(QPaintEvent *event) {
            QPainter painter(this);
            painter.setRenderHint(QPainter::Antialiasing);
            draw_timeline(painter);
            draw_time_ruler(painter);
            draw_playhead(painter);
        }

        void TimelineEditor::draw_timeline(QPainter &painter) {
            painter.fillRect(rect(), QColor(30, 30, 30));

            int y_start = 30;
            int height = 60;

            for (int i = 0; i < subtitles_.size(); ++i) {
                const auto& sub = subtitles_[i];
                int x1 = time_to_x(sub.start_time);
                int x2 = time_to_x(sub.end_time);
                int width = x2 - x1;

                QColor color = (i == selected_subtitle_id_) ? QColor(66, 153, 225) : QColor(72, 72, 72);
                painter.fillRect(x1, y_start, width, height, color);

                painter.setPen(QColor(200, 200, 200));
                painter.drawText(x1 + 4, y_start + 20, sub.text.left(30));
            }
        }

        void TimelineEditor::draw_time_ruler(QPainter &painter) {
            painter.fillRect(0, 0, width(), 25, QColor(40, 40, 40));
            painter.setPen(QColor(150, 150, 150));

            double step = 1.0 / zoom_;
            if (step < 0.1) step = 0.1;
            else if (step < 0.5) step = 0.5;
            else if (step < 1.0) step = 1.0;
            else if (step < 5.0) step = 5.0;
            else step = 10.0;

            for (double t = 0; t < duration_; t += step) {
                int x = time_to_x(t);
                painter.drawLine(x, 20, x, 25);

                int minutes = static_cast<int>(t) / 60;
                int seconds = static_cast<int>(t) % 60;
                painter.drawText(x + 2, 15, QString("%1:%2").arg(minutes, 2, 10, QChar('0')).arg(seconds, 2, 10, QChar('0')));
            }
        }

        void TimelineEditor::draw_playhead(QPainter &painter) {
            int x = time_to_x(current_time_);
            painter.setPen(QPen(QColor(255, 0, 0), 2));
            painter.drawLine(x, 0, x, height());
        }

        void TimelineEditor::mousePressEvent(QMouseEvent *event) {
            if (event->button() == Qt::LeftButton) {
                int idx = find_subtitle_at(event->pos().x());
                if (idx >= 0) {
                    selected_subtitle_id_ = idx;
                    is_dragging_ = true;
                    drag_start_x_ = event->pos().x();
                    drag_subtitle_id_ = idx;
                    emit subtitle_selected(idx);
                } else {
                    emit time_requested(x_to_time(event->pos().x()));
                }
                update();
            }
        }

        void TimelineEditor::mouseMoveEvent(QMouseEvent *event) {
            if (is_dragging_ && drag_subtitle_id_ >= 0) {
                double delta = x_to_time(event->pos().x() - drag_start_x_);
                auto& sub = subtitles_[drag_subtitle_id_];
                sub.start_time = std::max(0.0, sub.start_time + delta);
                sub.end_time = std::max(sub.start_time + 0.1, sub.end_time + delta);
                drag_start_x_ = event->pos().x();
                emit subtitle_changed(subtitles_);
                update();
            }
        }

        void TimelineEditor::mouseReleaseEvent(QMouseEvent *event) {
            if (event->button() == Qt::LeftButton) {
                is_dragging_ = false;
                drag_subtitle_id_ = -1;
            }
        }

        void TimelineEditor::wheelEvent(QWheelEvent *event) {
            if (event->angleDelta().y() > 0) {
                zoom_ = std::min(500.0, zoom_ * 1.2);
            } else {
                zoom_ = std::max(10.0, zoom_ / 1.2);
            }
            emit subtitle_changed(subtitles_);
            update();
        }

void ApiClient::upload_video(const QString& file_path) {
    QUrl url(base_url_ + "/api/videos/upload");
    QNetworkRequest request(url);

    QFile *file = new QFile(file_path);
    if (!file->open(QIODevice::ReadOnly)) {
        emit error("Cannot open file: " + file_path);
        return;
    }

    QHttpMultiPart *multi_part = new QHttpMultiPart(QHttpMultiPart::FormDataType);
    QHttpPart video_part;
    video_part.setHeader(QNetworkRequest::ContentTypeHeader, QVariant("video/mp4"));
    video_part.setHeader(QNetworkRequest::ContentDispositionHeader, QVariant("form-data; name=\"video\"; filename=\"" + file_path + "\""));
    video_part.setBodyDevice(file);
    file->setParent(multi_part);

    multi_part->append(video_part);

    QNetworkReply *reply = network_manager_->post(request, multi_part);
    multi_part->setParent(reply);
}

void ApiClient::export_video(const QString& video_id, const QVector<subforge::Subtitle>& subtitles) {
    QUrl url(base_url_ + "/api/tasks/export");
    QNetworkRequest request(url);
    request.setHeader(QNetworkRequest::ContentTypeHeader, "application/json");

    QJsonObject body;
    body["video_id"] = video_id;

    QJsonArray subs_array;
    for (const auto& sub : subtitles) {
        QJsonObject sub_obj;
        sub_obj["id"] = sub.id;
        sub_obj["start_time"] = sub.start_time;
        sub_obj["end_time"] = sub.end_time;
        sub_obj["text"] = QString::fromStdString(sub.text);
        subs_array.append(sub_obj);
    }
    body["subtitles"] = subs_array;

    QNetworkReply *reply = network_manager_->post(request, QJsonDocument(body).toJson());
}

void ApiClient::on_reply_finished(QNetworkReply* reply) {
    if (reply->error() != QNetworkReply::NoError) {
        emit error(reply->errorString());
        reply->deleteLater();
        return;
    }

    QByteArray data = reply->readAll();
    QJsonDocument doc = QJsonDocument::fromJson(data);

    if (reply->request().url().path().contains("/upload")) {
        QJsonObject obj = doc.object();
        emit upload_finished(obj["video_id"].toString());
    } else if (reply->request().url().path().contains("/transcribe")) {
        QJsonObject obj = doc.object();
        emit task_progress(0);
    } else if (reply->request().url().path().contains("/tasks/") && !reply->request().url().path().contains("/transcribe") && !reply->request().url().path().contains("/export")) {
        QJsonObject obj = doc.object();
        emit task_progress(obj["progress"].toInt());
        if (obj["status"].toString() == "done") {
            download_subtitles(obj["video_id"].toString());
        }
    } else if (reply->request().url().path().contains("/subtitles")) {
        QJsonArray arr = doc.object()["subtitles"].toArray();
        QVector<subforge::Subtitle> subtitles;
        for (const auto& val : arr) {
            QJsonObject obj = val.toObject();
            subforge::Subtitle sub;
            sub.id = obj["id"].toInt();
            sub.start_time = obj["start_time"].toDouble();
            sub.end_time = obj["end_time"].toDouble();
            sub.text = obj["text"].toString().toStdString();
            subtitles.append(sub);
        }
        emit subtitles_ready(subtitles);
    } else if (reply->request().url().path().contains("/export")) {
        QJsonObject obj = doc.object();
        emit export_finished(obj["file_path"].toString());
    }

    reply->deleteLater();
}

    std::uniform_int_distribution<> dis(0, 15);

void TranscribeHandler::handle_transcribe(const httplib::Request& req, httplib::Response& res) {
    try {
        json body = json::parse(req.body);
        std::string video_id = body.value("video_id", "");
        std::string language = body.value("language", "auto");

        if (video_id.empty()) {
            res.status = 400;
            json error_response = {{"error", "video_id is required"}};
            res.set_content(error_response.dump(), "application/json");
            return;
        }

        auto& config = Config::instance();
        std::string video_path = config.upload_dir() + "/" + video_id + ".mp4";
        std::string temp_dir = config.temp_dir() + "/" + video_id;

        std::string task_id = generate_task_id();
        task_progress_map[task_id] = 0;

        std::thread([video_path, temp_dir, language, task_id]() {
            try {
                task_progress_map[task_id] = 10;

                AudioExtractor extractor;
                std::string audio_path = extractor.extract(video_path, temp_dir);
                task_progress_map[task_id] = 40;

                Transcriber transcriber;
                auto subtitles = transcriber.transcribe(audio_path, language);
                task_progress_map[task_id] = 90;

                std::lock_guard<std::mutex> lock(task_mutex);
                task_progress_map[task_id] = 100;
            } catch (const std::exception& e) {
                std::lock_guard<std::mutex> lock(task_mutex);
                task_progress_map[task_id] = -1;
            }
        }).detach();

        json response = {
            {"task_id", task_id},
            {"status", "processing"},
            {"message", "Transcription started"}
        };
        res.set_content(response.dump(), "application/json");
    } catch (const std::exception& e) {
        res.status = 500;
        json error_response = {{"error", e.what()}};
        res.set_content(error_response.dump(), "application/json");
    }
}

void TranscribeHandler::handle_task_status(const httplib::Request& req, httplib::Response& res) {
    std::string task_id = req.path_params.at("task_id");

    std::lock_guard<std::mutex> lock(task_mutex);
    auto it = task_progress_map.find(task_id);
    if (it == task_progress_map.end()) {
        res.status = 404;
        json error_response = {{"error", "Task not found"}};
        res.set_content(error_response.dump(), "application/json");
        return;
    }

    int progress = it->second;
    std::string status = "processing";
    if (progress == 100) status = "done";
    else if (progress == -1) status = "error";

    json response = {
        {"task_id", task_id},
        {"progress", progress},
        {"status", status}
    };
    res.set_content(response.dump(), "application/json");
}

void ExportHandler::handle_export_video(const httplib::Request& req, httplib::Response& res) {
    try {
        json body = json::parse(req.body);
        std::string video_id = body.value("video_id", "");
        json subtitles_json = body.value("subtitles", json::array());

        if (video_id.empty()) {
            res.status = 400;
            json error_response = {{"error", "video_id is required"}};
            res.set_content(error_response.dump(), "application/json");
            return;
        }

        auto& config = Config::instance();
        std::string video_path = config.upload_dir() + "/" + video_id + ".mp4";
        std::string output_path = config.temp_dir() + "/" + video_id + "_output.mp4";

        std::string srt_path = config.temp_dir() + "/" + video_id + "/subtitles.srt";
        std::ofstream srt_file(srt_path);

        for (size_t i = 0; i < subtitles_json.size(); ++i) {
            const auto& sub = subtitles_json[i];
            srt_file << (i + 1) << "\n";

            auto format_time = [](double seconds) -> std::string {
                int hours = static_cast<int>(seconds) / 3600;
                int minutes = (static_cast<int>(seconds) % 3600) / 60;
                int secs = static_cast<int>(seconds) % 60;
                int ms = static_cast<int>((seconds - static_cast<int>(seconds)) * 1000);
                char buf[20];
                snprintf(buf, sizeof(buf), "%02d:%02d:%02d,%03d", hours, minutes, secs, ms);
                return buf;
            };

            srt_file << format_time(sub["start_time"].get<double>()) << " --> " << format_time(sub["end_time"].get<double>()) << "\n";
            srt_file << sub["text"].get<std::string>() << "\n\n";
        }
        srt_file.close();

        std::string ffmpeg_cmd = config.ffmpeg_path() + " -y -i \"" + video_path + "\" -vf \"subtitles=" + srt_path + "\" -c:a copy \"" + output_path + "\" 2>&1";
        std::array<char, 128> buffer;
        std::string result;
        std::unique_ptr<FILE, decltype(&pclose)> pipe(popen(ffmpeg_cmd.c_str(), "r"), pclose);
        if (!pipe) throw std::runtime_error("popen() failed!");
        while (fgets(buffer.data(), buffer.size(), pipe.get()) != nullptr) {
            result += buffer.data();
        }

        json response = {
            {"file_path", output_path},
            {"message", "Export successful"}
        };
        res.set_content(response.dump(), "application/json");
    } catch (const std::exception& e) {
        res.status = 500;
        json error_response = {{"error", e.what()}};
        res.set_content(error_response.dump(), "application/json");
    }
}

} // namespace subforge

StyleEditorWidget::StyleEditorWidget(QWidget *parent) : QWidget(parent) {
    setup_ui();
}

void StyleEditorWidget::setup_ui() {
    auto layout = new QFormLayout(this);

    font_combo_ = new QFontComboBox(this);
    layout->addRow("Font:", font_combo_);

    font_size_spin_ = new QSpinBox(this);
    font_size_spin_->setRange(8, 72);
    font_size_spin_->setValue(24);
    layout->addRow("Size:", font_size_spin_);

    position_combo_ = new QComboBox(this);
    position_combo_->addItems({"bottom_center", "top_center", "middle_center"});
    layout->addRow("Position:", position_combo_);

    bold_check_ = new QCheckBox("Bold", this);
    italic_check_ = new QCheckBox("Italic", this);
    auto style_layout = new QHBoxLayout();
    style_layout->addWidget(bold_check_);
    style_layout->addWidget(italic_check_);
    layout->addRow("Style:", style_layout);

    color_button_ = new QPushButton("Text Color", this);
    bg_color_button_ = new QPushButton("Background Color", this);
    auto color_layout = new QHBoxLayout();
    color_layout->addWidget(color_button_);
    color_layout->addWidget(bg_color_button_);
    layout->addRow("Colors:", color_layout);

    connect(font_combo_, &QFontComboBox::currentFontChanged, this, &StyleEditorWidget::on_style_changed);
    connect(font_size_spin_, QOverload<int>::of(&QSpinBox::valueChanged), this, &StyleEditorWidget::on_style_changed);
    connect(position_combo_, QOverload<int>::of(&QComboBox::currentIndexChanged), this, &StyleEditorWidget::on_style_changed);
    connect(bold_check_, &QCheckBox::toggled, this, &StyleEditorWidget::on_style_changed);
    connect(italic_check_, &QCheckBox::toggled, this, &StyleEditorWidget::on_style_changed);
    connect(color_button_, &QPushButton::clicked, this, [this]() {
        QColor color = QColorDialog::getColor(QColor(QString::fromStdString(current_style_.color)), this);
        if (color.isValid()) {
            current_style_.color = color.name().toStdString();
            emit style_changed(current_style_);
        }
    });
    connect(bg_color_button_, &QPushButton::clicked, this, [this]() {
        QColor color = QColorDialog::getColor(QColor(QString::fromStdString(current_style_.bg_color)), this);
        if (color.isValid()) {
            current_style_.bg_color = color.name(QColor::HexArgb).toStdString();
            emit style_changed(current_style_);
        }
    });
}

void StyleEditorWidget::set_style(const subforge::SubtitleStyle& style) {
    current_style_ = style;
    font_combo_->setCurrentFont(QFont(QString::fromStdString(style.font_family)));
    font_size_spin_->setValue(style.font_size);
    bold_check_->setChecked(style.bold);
    italic_check_->setChecked(style.italic);

    int pos_idx = position_combo_->findText(QString::fromStdString(style.position));
    if (pos_idx >= 0) position_combo_->setCurrentIndex(pos_idx);
}

subforge::SubtitleStyle StyleEditorWidget::get_style() const {
    return current_style_;
}

void StyleEditorWidget::on_style_changed() {
    current_style_.font_family = font_combo_->currentFont().family().toStdString();
    current_style_.font_size = font_size_spin_->value();
    current_style_.bold = bold_check_->isChecked();
    current_style_.italic = italic_check_->isChecked();
    current_style_.position = position_combo_->currentText().toStdString();

    emit style_changed(current_style_);
}
